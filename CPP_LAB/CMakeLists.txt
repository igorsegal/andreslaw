cmake_minimum_required(VERSION 3.16)
project(andreslaw_cpp_lab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(as_core INTERFACE)
target_include_directories(as_core INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_executable(as_lab src/main.cpp)
target_link_libraries(as_lab PRIVATE as_core)
add_executable(as_tests tests/test_contracts.cpp)
target_link_libraries(as_tests PRIVATE as_core)
enable_testing()
add_test(NAME as_contracts COMMAND as_tests)
add_executable(as_sim_tests tests/test_simulator.cpp)
target_link_libraries(as_sim_tests PRIVATE as_core)
add_test(NAME as_simulator_contracts COMMAND as_sim_tests)
